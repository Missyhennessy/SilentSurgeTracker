```text-sm font-ligatures-none
import numpy as np

def breakout_probability(sss, velocity, sentiment, anchor):
    z = 0.05 * sss + 0.3 * velocity + 0.4 * sentiment + 0.2 * anchor - 10
    return round(1 / (1 + np.exp(-z)) * 100, 2)

```

### 📊 5\. `surge_engine.py`

```text-sm font-ligatures-none
import pandas as pd
from forecast_model import breakout_probability
from decision_logic import suggest_action

# Load data
df = pd.read_csv("data/token_metrics.csv")

# Apply model
df["Breakout_3d"] = df.apply(lambda row: breakout_probability(row["SSS"], row["Velocity"], row["Sentiment"], row["AnchorPressure"] * 0.8), axis=1)
df["Breakout_7d"] = df.apply(lambda row: breakout_probability(row["SSS"], row["Velocity"], row["Sentiment"], row["AnchorPressure"]), axis=1)

# Decision logic
df["SuggestedAction"] = df.apply(lambda row: suggest_action(row["Breakout_3d"], row["Breakout_7d"], row["SSS"], row["AnchorPressure"]), axis=1)

# Output
df_sorted = df.sort_values(by="Breakout_7d", ascending=False)
print(df_sorted[["Token", "Breakout_3d", "Breakout_7d", "SuggestedAction"]])

```
